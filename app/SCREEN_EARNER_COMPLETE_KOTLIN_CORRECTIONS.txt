package com.example.data.model

import com.example.data.engine.CurrencyManager
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

enum class BalanceDetectionStatus {
    SCANNING,
    DETECTED,
    VERIFIED,
    UNAVAILABLE,
    STALE
}

data class BalanceState(
    val status: BalanceDetectionStatus = BalanceDetectionStatus.SCANNING,
    val amount: Double? = null,
    val currency: String? = null,
    val lastReadTimestampMs: Long = 0L,
    val failureReason: String? = null
) {
    val lastUpdatedMs: Long
        get() = lastReadTimestampMs

    val isUsableForBet: Boolean
        get() = (status == BalanceDetectionStatus.VERIFIED || status == BalanceDetectionStatus.DETECTED) &&
            amount != null &&
            amount.isFinite() &&
            amount >= 0.0

    val isStale: Boolean
        get() = status == BalanceDetectionStatus.STALE

    fun formattedDisplay(): String {
        return when (status) {
            BalanceDetectionStatus.VERIFIED -> {
                val amt = amount ?: 0.0
                "${CurrencyManager.formatAmount(amt)} [VERIFIED]"
            }
            BalanceDetectionStatus.DETECTED -> {
                val amt = amount ?: 0.0
                "${CurrencyManager.formatAmount(amt)} [DETECTED]"
            }
            BalanceDetectionStatus.SCANNING -> "SCANNING..."
            BalanceDetectionStatus.UNAVAILABLE -> "UNAVAILABLE${failureReason?.let { " ($it)" } ?: ""}"
            BalanceDetectionStatus.STALE -> "STALE (Needs refresh)"
        }
    }
}

enum class ActionExecutionState {
    IDLE,
    SCANNING,
    READING_BALANCE,
    DETECTING_CONTROLS,
    VALIDATING_GAME_STATE,
    EXECUTING_ACTION,
    VERIFYING_ACTION,
    COMPLETED,
    FAILED,
    PAUSED
}

data class ActionExecutionRecord(
    val actionId: String = "act_${System.currentTimeMillis()}_${(1000..9999).random()}",
    val operationName: String,
    val state: ActionExecutionState = ActionExecutionState.IDLE,
    val startTimestampMs: Long = System.currentTimeMillis(),
    val completionTimestampMs: Long? = null,
    val targetCoordinates: Pair<Float, Float>? = null,
    val isSuccess: Boolean? = null,
    val failureReason: String? = null,
    val gestureDispatched: Boolean = false,
    val interfaceResponded: Boolean = false,
    val stateConfirmed: Boolean = false,
    val details: String = ""
) {
    val durationMs: Long
        get() = (completionTimestampMs ?: System.currentTimeMillis()) - startTimestampMs
}

package com.example.service

import android.util.Log
import android.view.accessibility.AccessibilityEvent
import com.example.data.engine.GameDetection
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow

class ScreenEarnerAccessibilityService : AccessibilityController() {

    companion object {
        private const val TAG = "ScreenEarnerA11ySvc"
        var instance: ScreenEarnerAccessibilityService? = null
            private set

        private val _connectionStatus = MutableStateFlow(AccessibilityConnectionStatus.DISCONNECTED)
        val connectionStatus: StateFlow<AccessibilityConnectionStatus> = _connectionStatus.asStateFlow()

        fun isServiceConnected(): Boolean =
            _connectionStatus.value == AccessibilityConnectionStatus.CONNECTED && instance != null

        fun markConnecting() {
            if (_connectionStatus.value != AccessibilityConnectionStatus.CONNECTED) {
                _connectionStatus.value = AccessibilityConnectionStatus.CONNECTING
            }
        }

        private val _detectionState = MutableStateFlow(GameDetectionState())
        val detectionState: StateFlow<GameDetectionState> = _detectionState.asStateFlow()
    }

    override fun onServiceConnected() {
        try {
            super.onServiceConnected()
            instance = this
            _connectionStatus.value = AccessibilityConnectionStatus.CONNECTED
            Log.i(TAG, "ScreenEarnerAccessibilityService connected successfully.")
        } catch (t: Throwable) {
            _connectionStatus.value = AccessibilityConnectionStatus.ERROR
            Log.e(TAG, "Error in ScreenEarnerAccessibilityService.onServiceConnected", t)
        }
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        try {
            if (event == null) return
            val packageName = try { event.packageName?.toString().orEmpty() } catch (_: Throwable) { "" }
            val ownPkg = runCatching { applicationContext?.packageName }.getOrNull()
            if (!ownPkg.isNullOrBlank() && packageName == ownPkg) return

            super.onAccessibilityEvent(event)
            // Evaluates and updates GameDetectionState safely
        } catch (t: Throwable) {
            Log.e(TAG, "Caught non-fatal exception in onAccessibilityEvent", t)
        }
    }

    override fun onInterrupt() {
        try {
            super.onInterrupt()
        } catch (t: Throwable) {
            _connectionStatus.value = AccessibilityConnectionStatus.ERROR
            Log.e(TAG, "Error in onInterrupt", t)
        }
    }

    override fun onUnbind(intent: android.content.Intent?): Boolean {
        return try {
            if (instance == this) instance = null
            _connectionStatus.value = AccessibilityConnectionStatus.DISCONNECTED
            super.onUnbind(intent)
        } catch (t: Throwable) {
            _connectionStatus.value = AccessibilityConnectionStatus.ERROR
            Log.e(TAG, "Error in onUnbind", t)
            false
        }
    }

    override fun onDestroy() {
        try {
            super.onDestroy()
            if (instance == this) instance = null
            _connectionStatus.value = AccessibilityConnectionStatus.DISCONNECTED
        } catch (t: Throwable) {
            _connectionStatus.value = AccessibilityConnectionStatus.ERROR
            Log.e(TAG, "Error in onDestroy", t)
        }
    }
}
